const page = new URL(location.href);
const id = page.searchParams.get('v');
if (page.origin !== 'https://www.youtube.com' || page.username || page.password || page.pathname !== '/watch' ||
    !/^[A-Za-z0-9_-]{11}$/.test(id || '')) {
    throw new Error('请在 YouTube 单视频观看页使用');
}
if (player.presetEvent.syncTime) {
    const video = document.querySelector('#movie_player video.html5-main-video');
    if (!video || video.readyState < 1 || !Number.isFinite(video.duration) ||
        !Number.isFinite(video.currentTime) || document.querySelector('#movie_player.ad-showing')) {
        throw new Error('请等待正片开始播放');
    }
    page.search = new URLSearchParams({v: id, t: Math.floor(video.currentTime)});
    page.hash = '';
}
window.open('ush://MPVNET?' + compress('"' + page.href + '"'), '_self');
